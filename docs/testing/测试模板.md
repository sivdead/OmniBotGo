```go
package entity

import (
    "testing"
    "time"
    
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/require"
)

// 基础实体测试模板
func TestEntityName_BasicOperations(t *testing.T) {
    entity := EntityName{
        Field1: "value1",
        Field2: 123,
        Field3: true,
    }
    
    // 验证字段值
    assert.Equal(t, "value1", entity.Field1)
    assert.Equal(t, 123, entity.Field2)
    assert.True(t, entity.Field3)
}

// 验证方法测试模板
func TestEntityName_Validate(t *testing.T) {
    tests := []struct {
        name    string
        entity  EntityName
        wantErr bool
        errMsg  string
    }{
        {
            name: "valid entity",
            entity: EntityName{
                RequiredField: "value",
                OptionalField: "optional",
            },
            wantErr: false,
        },
        {
            name: "missing required field",
            entity: EntityName{
                OptionalField: "optional",
            },
            wantErr: true,
            errMsg:  "required field is missing",
        },
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            err := tt.entity.Validate()
            
            if tt.wantErr {
                assert.Error(t, err)
                if tt.errMsg != "" {
                    assert.Contains(t, err.Error(), tt.errMsg)
                }
            } else {
                assert.NoError(t, err)
            }
        })
    }
}

// JSONField测试模板
func TestEntityName_JSONField(t *testing.T) {
    entity := EntityName{
        Config: JSONField{
            "string_field": "test_value",
            "int_field":    42,
            "bool_field":   true,
            "nested_field": map[string]interface{}{
                "sub_field": "sub_value",
            },
        },
    }
    
    // 测试基础类型获取
    assert.Equal(t, "test_value", entity.Config.GetString("string_field"))
    assert.Equal(t, 42, entity.Config.GetInt("int_field"))
    assert.True(t, entity.Config.GetBool("bool_field"))
    
    // 测试不存在字段的默认值
    assert.Empty(t, entity.Config.GetString("nonexistent"))
    assert.Equal(t, 0, entity.Config.GetInt("nonexistent"))
    assert.False(t, entity.Config.GetBool("nonexistent"))
    
    // 测试序列化
    value, err := entity.Config.Value()
    assert.NoError(t, err)
    assert.NotNil(t, value)
}

func TestEntity_Validation(t *testing.T) {
    tests := []struct {
        name    string
        entity  Entity
        wantErr bool
    }{
        {"valid", Entity{Field: "value"}, false},
        {"invalid", Entity{}, true},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            err := tt.entity.Validate()
            assert.Equal(t, tt.wantErr, err != nil)
        })
    }
}
```

```go
package usecase_test

import (
    "context"
    "testing"

    suite.Suite
    ctrl        *gomock.Controller
    usecase     *usecase.UseCaseName
    mockRepo    *MockRepository
    mockService *MockService
    logger      *zerolog.Logger
}

func (s *UseCaseNameTestSuite) SetupTest() {
    s.ctrl = gomock.NewController(s.T())
    s.mockRepo = NewMockRepository(s.ctrl)
    s.mockService = NewMockService(s.ctrl)
    
    logger := zerolog.Nop()
    s.logger = &logger
    
    s.usecase = usecase.NewUseCaseName(
        s.mockRepo,
        s.mockService,
        s.logger,
    )
}

func (s *UseCaseNameTestSuite) TearDownTest() {
    s.ctrl.Finish()
}

// 成功场景测试模板
func (s *UseCaseNameTestSuite) TestMethodName_Success() {
    // Arrange - 准备测试数据
    input := &dto.InputDTO{
        Field1: "value1",
        Field2: 123,
    }
    
    expectedEntity := &entity.EntityName{
        Field1: input.Field1,
        Field2: input.Field2,
        Status: entity.StatusActive,
    }
    
    // 设置Mock期望
    s.mockRepo.EXPECT().
        Create(gomock.Any(), gomock.Eq(expectedEntity)).
        Return(nil).
        Times(1)
    
    s.mockService.EXPECT().
        ProcessData(gomock.Any(), gomock.Any()).
        Return(nil).
        Times(1)
    
    // Act - 执行被测试的方法
    result, err := s.usecase.MethodName(context.Background(), input)
    
    // Assert - 验证结果
    s.NoError(err)
    s.NotNil(result)
    s.Equal(expectedEntity.Field1, result.Field1)
}

// 错误场景测试模板
```

```go
package persistent_test

import (
    "context"
    "testing"
    
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/require"
    "gorm.io/driver/sqlite"

// Repository测试模板
func TestRepositoryName_CRUD(t *testing.T) {
    // 设置内存数据库
    db := setupTestDB(t)
    defer cleanupTestDB(db)
    
    repo := persistent.NewRepositoryName(db)
    ctx := context.Background()
    
    t.Run("Create", func(t *testing.T) {
        entity := &entity.EntityName{
            Field1: "test_value",
            Field2: 123,
            Status: entity.StatusActive,
        }
        
        err := repo.Create(ctx, entity)
        assert.NoError(t, err)
        assert.NotZero(t, entity.ID)
        assert.False(t, entity.CreatedAt.IsZero())
    })
    
    t.Run("GetByID", func(t *testing.T) {
        // 先创建一个实体
        entity := &entity.EntityName{
            Field1: "test_value",
            Field2: 123,
        }
        err := repo.Create(ctx, entity)
        require.NoError(t, err)
        
        // 查询实体
        found, err := repo.GetByID(ctx, entity.ID)
        assert.NoError(t, err)
        assert.NotNil(t, found)
        assert.Equal(t, entity.Field1, found.Field1)
        assert.Equal(t, entity.Field2, found.Field2)
    })
    
    t.Run("Update", func(t *testing.T) {
        // 创建实体
        entity := &entity.EntityName{
            Field1: "original_value",
            Field2: 123,
        }
        err := repo.Create(ctx, entity)
        require.NoError(t, err)
        
        // 更新实体
        entity.Field1 = "updated_value"
        err = repo.Update(ctx, entity)
        assert.NoError(t, err)
        
        // 验证更新
        found, err := repo.GetByID(ctx, entity.ID)
        assert.NoError(t, err)
        assert.Equal(t, "updated_value", found.Field1)
    })
    
    t.Run("Delete", func(t *testing.T) {
        // 创建实体
        entity := &entity.EntityName{
            Field1: "test_value",
            Field2: 123,
        }
        err := repo.Create(ctx, entity)
        require.NoError(t, err)
        
        // 删除实体
        err = repo.Delete(ctx, entity.ID)
        assert.NoError(t, err)
        
        // 验证删除
        found, err := repo.GetByID(ctx, entity.ID)
        assert.Error(t, err)
        assert.Nil(t, found)
    })
    
    t.Run("List", func(t *testing.T) {
        // 创建多个实体
        entities := []*entity.EntityName{
            {Field1: "value1", Field2: 1},
            {Field1: "value2", Field2: 2},
            {Field1: "value3", Field2: 3},
        }
        
        for _, e := range entities {
            err := repo.Create(ctx, e)
            require.NoError(t, err)
        }
        
        // 查询列表
        result, err := repo.List(ctx, map[string]interface{}{
            "limit":  10,
            "offset": 0,
        })
        
        assert.NoError(t, err)
        assert.GreaterOrEqual(t, len(result), 3)
    })
}

// 测试数据库设置
func setupTestDB(t *testing.T) *gorm.DB {
    db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{})
    require.NoError(t, err)
    
    // 运行迁移
    err = db.AutoMigrate(&entity.EntityName{})
    require.NoError(t, err)
    
    return db
}

func cleanupTestDB(db *gorm.DB) {
    sqlDB, _ := db.DB()
    sqlDB.Close()
}
```

```go
package v1_test

import (
    "bytes"

// Controller测试模板
func TestControllerName_Method(t *testing.T) {
    // 设置Mock
    ctrl := gomock.NewController(t)
    defer ctrl.Finish()
    
    mockUseCase := NewMockUseCase(ctrl)
    logger := zerolog.Nop()
    
    controller := v1.NewControllerName(mockUseCase, &logger)
    
    // 创建Fiber应用
    app := fiber.New()
    controller.RegisterRoutes(app)
    
    t.Run("Success", func(t *testing.T) {
        // 准备请求数据
        payload := map[string]interface{}{
            "field1": "value1",
            "field2": 123,
            "field3": true,
        }
        
        body, err := json.Marshal(payload)
        require.NoError(t, err)
        
        // 设置Mock期望
        expectedDTO := &dto.InputDTO{
            Field1: "value1",
            Field2: 123,
            Field3: true,
        }
        
        expectedResult := &dto.OutputDTO{
            ID:     1,
            Status: "success",
        }
        
        mockUseCase.EXPECT().
            ProcessData(gomock.Any(), gomock.Eq(expectedDTO)).
            Return(expectedResult, nil).
            Times(1)
        
        // 发送请求
        req := httptest.NewRequest("POST", "/api/v1/resource", bytes.NewReader(body))
        req.Header.Set("Content-Type", "application/json")
        
        resp, err := app.Test(req)
        require.NoError(t, err)
        
        // 验证响应
        assert.Equal(t, http.StatusOK, resp.StatusCode)
        
        var response map[string]interface{}
        err = json.NewDecoder(resp.Body).Decode(&response)
        require.NoError(t, err)
        
        assert.True(t, response["success"].(bool))
        assert.NotNil(t, response["data"])
    })
    
    t.Run("BadRequest", func(t *testing.T) {
        // 发送无效请求
        req := httptest.NewRequest("POST", "/api/v1/resource", bytes.NewReader([]byte("invalid json")))
        req.Header.Set("Content-Type", "application/json")
        
        resp, err := app.Test(req)
        require.NoError(t, err)
        
        assert.Equal(t, http.StatusBadRequest, resp.StatusCode)
        
        var response map[string]interface{}
        err = json.NewDecoder(resp.Body).Decode(&response)
        require.NoError(t, err)
        
        assert.False(t, response["success"].(bool))
        assert.NotEmpty(t, response["error"])
    })
    
    t.Run("InternalServerError", func(t *testing.T) {
        payload := map[string]interface{}{
            "field1": "value1",
            "field2": 123,
        }
        
        body, err := json.Marshal(payload)
        require.NoError(t, err)
        
        // 设置Mock返回错误
        mockUseCase.EXPECT().
            ProcessData(gomock.Any(), gomock.Any()).
            Return(nil, errors.New("internal error")).
            Times(1)
        
        req := httptest.NewRequest("POST", "/api/v1/resource", bytes.NewReader(body))
        req.Header.Set("Content-Type", "application/json")
        
        resp, err := app.Test(req)
        require.NoError(t, err)
        
        assert.Equal(t, http.StatusInternalServerError, resp.StatusCode)
    })
}
```

```go
package integration_test

type IntegrationTestSuite struct {
    suite.Suite
    app     *app.App
    baseURL string
    client  *http.Client
}

func (s *IntegrationTestSuite) SetupSuite() {
    // 创建测试配置
    cfg := &config.Config{
        DB: config.DB{
            Type: "sqlite",
            DSN:  ":memory:",
        },
        HTTP: config.HTTP{
            Port: "0", // 随机端口
        },
    }

    s.client = &http.Client{Timeout: 10 * time.Second}
    
    // 启动应用
    go func() {
        err := s.app.HTTPServer.Listen(":" + cfg.HTTP.Port)
        if err != nil {
            s.T().Logf("HTTP server error: %v", err)
        }
    }()
    
    // 等待服务启动
    time.Sleep(100 * time.Millisecond)
}

func (s *IntegrationTestSuite) TearDownSuite() {
    if s.app != nil {
        s.app.HTTPServer.Shutdown()
        if s.app.Database != nil {
            s.app.Database.Close()
        }
    }
}

func (s *IntegrationTestSuite) TestCompleteWorkflow() {
    // 1. 创建机器人
    bot := s.createBot()
    
    // 2. 创建通道
    channel := s.createChannel(bot.ID)
    
    // 3. 发送消息
    message := s.sendMessage(channel.ID)
    
    // 4. 查询消息
    s.verifyMessage(message.ID)

func (s *IntegrationTestSuite) createBot() *BotResponse {
    payload := map[string]interface{}{
        "bot_name":    "集成测试机器人",
        "bot_type":    "general",
        "description": "用于集成测试",
        "created_by":  "test",
    }
    
    body, err := json.Marshal(payload)
    s.Require().NoError(err)
    
    resp, err := s.client.Post(
        s.baseURL+"/api/v1/bots",
        "application/json",
        bytes.NewReader(body),
    )
    s.Require().NoError(err)
    defer resp.Body.Close()
    
    s.Assert().Equal(http.StatusCreated, resp.StatusCode)
    
    var response struct {
        Success bool         `json:"success"`
        Data    *BotResponse `json:"data"`

    err = json.NewDecoder(resp.Body).Decode(&response)
    s.Require().NoError(err)
    s.Assert().True(response.Success)
    
    return response.Data
}
```

```go
func TestConcurrentOperations(t *testing.T) {
    const (
        numGoroutines = 100
        operationsPerGoroutine = 10
    )
    
    usecase := setupTestUseCase(t)
    
    var wg sync.WaitGroup
    errors := make(chan error, numGoroutines*operationsPerGoroutine)
    
    for i := 0; i < numGoroutines; i++ {
        wg.Add(1)
        go func(goroutineID int) {
            defer wg.Done()
            
            for j := 0; j < operationsPerGoroutine; j++ {
                input := &dto.InputDTO{
                    Field1: fmt.Sprintf("concurrent_%d_%d", goroutineID, j),
                    Field2: goroutineID*1000 + j,
                }
                
                _, err := usecase.ProcessData(context.Background(), input)
                if err != nil {
                    errors <- fmt.Errorf("goroutine %d, operation %d: %w", goroutineID, j, err)
                    return
                }
            }
        }(i)
    }
    
    wg.Wait()
    close(errors)
    
    // 检查错误
    var errorCount int
    for err := range errors {
        t.Errorf("并发测试错误: %v", err)
        errorCount++
    }
    
    if errorCount > 0 {
        t.Fatalf("并发测试失败，共 %d 个错误", errorCount)
    }
    
    t.Logf("并发测试成功: %d 个goroutine，每个执行 %d 次操作", numGoroutines, operationsPerGoroutine)
}
```

```go
func CleanupTestDB(t *testing.T, db *gorm.DB) {
    t.Cleanup(func() {
        // 按依赖关系逆序清理
        tables := []string{
            "api_call_logs",
            "connection_logs", 
            "message_routing_rules",
            "message_processors",
            "message_queues",
            "messages",
            "channels",
            "bots",
            "system_configs",
        }

        }

    })
```
