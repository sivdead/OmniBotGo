# OmniBotGo 测试执行指南

本指南提供了OmniBotGo项目测试的具体执行方法和最佳实践。

## 快速开始

### 环境准备
```bash
# 1. 安装测试依赖
go mod download

# 2. 启动测试数据库（可选）
just compose-up

# 3. 生成Mock文件
just mock

# 4. 运行所有测试
just test
```

### 基础测试命令
```bash
# 单元测试
go test ./internal/...

# 集成测试
go test ./integration-test/...

# 架构测试
go test ./test/architecture/...

# 带覆盖率的测试
go test -cover ./internal/...

# 详细测试输出
go test -v ./internal/...
```

// internal/repo/persistent/message_test.go
func TestMessageRepository_Integration(t *testing.T) {
    // 启动MySQL容器
    ctx := context.Background()
    req := testcontainers.ContainerRequest{
        Image:        "mysql:8.0",
        ExposedPorts: []string{"3306/tcp"},
        Env: map[string]string{
            "MYSQL_ROOT_PASSWORD": "password",
            "MYSQL_DATABASE":      "testdb",
        },
        WaitingFor: wait.ForLog("ready for connections"),
    }
    
    container, err := testcontainers.GenericContainer(ctx, testcontainers.GenericContainerRequest{
        ContainerRequest: req,
        Started:          true,
    })
    require.NoError(t, err)
    defer container.Terminate(ctx)
    
    // 获取数据库连接
    host, err := container.Host(ctx)
    require.NoError(t, err)
    port, err := container.MappedPort(ctx, "3306")
    require.NoError(t, err)
    
    dsn := fmt.Sprintf("root:password@tcp(%s:%s)/testdb?charset=utf8mb4&parseTime=True&loc=Local",
        host, port.Port())
    
    db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{})
    require.NoError(t, err)
    
    // 运行迁移
    err = db.AutoMigrate(&entity.Message{})
    require.NoError(t, err)
    
    // 创建repository并测试
    repo := NewMessageRepository(db)
    
    t.Run("Create and Get Message", func(t *testing.T) {
        msg := &entity.Message{
            MessageID:   "test_msg_001",
            ChannelID:   1,
            Direction:   entity.MessageDirectionInbound,
            MessageType: "text",
            Content:     "test message",
            Status:      entity.MessageStatusPending,
        }
        
        err := repo.Create(ctx, msg)
        assert.NoError(t, err)
        
        retrieved, err := repo.GetByID(ctx, msg.MessageID)
        assert.NoError(t, err)
        assert.Equal(t, msg.Content, retrieved.Content)
    })
}

func TestMessageController_SendMessage(t *testing.T) {
    // 创建Mock UseCase
    ctrl := gomock.NewController(t)
    defer ctrl.Finish()
    
    mockUseCase := mock.NewMockMessageUseCase(ctrl)
    logger := zerolog.Nop()

    // 创建Fiber应用
    app := fiber.New()

    t.Run("成功发送消息", func(t *testing.T) {
        // 准备请求数据
        payload := map[string]interface{}{
            "channel_id":    1,
            "message_type":  "text",
            "content":       "测试消息",
            "receiver_id":   "user123",
            "receiver_type": "user",
        }
        
        body, _ := json.Marshal(payload)
        
        // 设置Mock期望
        mockUseCase.EXPECT().
            SendMessage(gomock.Any(), gomock.Any()).
            Return(nil)
        
        // 发送请求
        req := httptest.NewRequest("POST", "/api/v1/messages", bytes.NewReader(body))
        req.Header.Set("Content-Type", "application/json")
        
        resp, err := app.Test(req)
        require.NoError(t, err)
        
        // 验证响应
        assert.Equal(t, http.StatusOK, resp.StatusCode)
        
        var response map[string]interface{}
        err = json.NewDecoder(resp.Body).Decode(&response)
        require.NoError(t, err)
        assert.True(t, response["success"].(bool))
    })
}

### 2. 负载测试

#### 并发消息处理测试
```go
func TestConcurrentMessageProcessing(t *testing.T) {
    const numGoroutines = 100
    const messagesPerGoroutine = 10
    
    usecase := setupTestUseCase(t)
    
    var wg sync.WaitGroup
    errors := make(chan error, numGoroutines*messagesPerGoroutine)
    
    for i := 0; i < numGoroutines; i++ {
        wg.Add(1)
        go func(goroutineID int) {
            defer wg.Done()
            
            for j := 0; j < messagesPerGoroutine; j++ {
                message := &dto.UnifiedMessage{
                    MessageID: fmt.Sprintf("concurrent_%d_%d", goroutineID, j),
                    Content:   fmt.Sprintf("Message from goroutine %d, iteration %d", goroutineID, j),
                }
                
                err := usecase.ProcessInboundMessage(context.Background(), message)
                if err != nil {
                    errors <- err
                    return
                }
            }
        }(i)
    }
    
    wg.Wait()
    close(errors)
    
    // 检查是否有错误
    for err := range errors {
        t.Errorf("并发测试失败: %v", err)
    }
}
```

### 1. 测试命名规范

### 2. 断言最佳实践
```go
// 使用具体的断言方法
assert.Equal(t, expected, actual)           // 而不是 assert.True(t, expected == actual)
assert.NoError(t, err)                     // 而不是 assert.Nil(t, err)
assert.Contains(t, slice, item)            // 而不是手动遍历

// 提供有意义的错误消息
assert.Equal(t, expected, actual, "用户ID应该匹配")
```

### 3. Mock使用原则
```go
// 只Mock外部依赖
mockRepo.EXPECT().Create(gomock.Any(), gomock.Any()).Return(nil)

// 使用具体的参数匹配
mockRepo.EXPECT().
    Create(gomock.Any(), gomock.Eq(expectedMessage)).
    Return(nil)

// 验证调用次数
mockRepo.EXPECT().Create(gomock.Any(), gomock.Any()).Times(1)
```

### 4. 测试隔离
```go
// 每个测试都应该独立
func TestSomething(t *testing.T) {
    // Setup - 创建测试所需的状态
    setup := createTestSetup(t)
    defer cleanup(setup) // 确保清理
    
    // Execute - 执行被测试的操作
    result := setup.service.DoSomething()
    
    // Verify - 验证结果
    assert.Equal(t, expected, result)
}
```
